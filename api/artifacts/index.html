    async def iter_results(
    self, *, successes_only: bool = False
) -> AsyncIterator[WorkItemResult[Any, Any]]:
    """Stream a finite, bounded-memory snapshot of complete item records.

    Raises ``ArtifactIOError`` if the store is closed, including when it is
    closed during iteration (checked before each page). Use
    ``JsonlArtifactStore.read_results(path)`` to read a closed artifact.
    """
    if not self._prepared:
        try:
            exists_and_nonempty = await asyncio.to_thread(
                lambda: self.path.exists() and self.path.stat().st_size > 0
            )
        except OSError as exc:
            raise ArtifactIOError(f"Could not inspect artifact {self.path}: {exc}") from exc
        if not exists_and_nonempty:
            # Preserve the historical create-on-first-operation behavior
            # for an explicitly identified empty store.
            await self._prepare()

    async def capture_snapshot() -> int:
        async with self._lock:
            self._raise_detached_io_error()
            # Closed means closed, as for every other operation and for
            # SqliteArtifactStore; read_results(path) reads a closed artifact.
            if self._closed:
                raise ArtifactIOError(f"Artifact store is closed: {self.path}")
            return await asyncio.to_thread(_artifact_snapshot_size, self.path)

    snapshot_size = await self._run_lock_owner(capture_snapshot())
    offset = 0
    line_number = 0
    manifest_seen = False
    done = False
    while not done:
        # A running iterator stops at its next page once the store closes,
        # as SqliteArtifactStore's does. Pages are read by path without the
        # lock; close() sets _closed on the loop, so this check can't tear.
        if self._closed:
            raise ArtifactIOError(f"Artifact store is closed: {self.path}")
        page, offset, line_number, manifest_seen, done = await asyncio.to_thread(
            _read_artifact_page,
            self.path,
            offset=offset,
            snapshot_size=snapshot_size,
            line_number=line_number,
            manifest_seen=manifest_seen,
            page_size=_JSONL_ITER_PAGE_SIZE,
        )
        for record in page:
            if successes_only and not record.get("success"):
                continue
            yield _decode_stored_record(
                record,
                sequence=_jsonl_display_sequence(record),
                output_decoder=self.output_decoder,
                context_decoder=self.context_decoder,
            )

    SqliteArtifactStore

    Indexed SQLite backend for large restartable runs (v0.21). Note that SqliteArtifactStore.read_results() is an async classmethod, unlike the synchronous JSONL convenience. It materializes a finite high-water snapshot through an operationally read-only path connection. Normal reads participate in SQLite locking and may create empty WAL/SHM coordination sidecars so a writer can safely start during inspection; live store.iter_results() instead flushes and inspects that writable store.

    async def close(self) -> None:

    async def lookup(
    self,
    work_item: LLMWorkItem[Any, Any, Any],
    prepared_item: Any,
    policy: ResumePolicy,
) -> WorkItemResult[Any, Any] | None:
    """Look up the newest compatible terminal row through a replay index."""
    if policy is ResumePolicy.NONE:
        return None
    await self._before_operation()
    prepared = self._coerce_prepared(prepared_item)
    await self._prepare()
    _, _, identity_fingerprint = self._require_resolved_identity()
    row = await self._run_db(
        self._lookup_sync,
        work_item.item_id,
        prepared,
        identity_fingerprint,
        policy,
    )
    if row is None or is_non_replayable_record(row):
        return None
    return _restore_store_record(
        self, row, work_item, location=f" at sequence {row.get('record_sequence')!r}"
    )

    self._resolve_identity_from(work_item.strategy)
    await self._prepare()
    _, _, fingerprint = self._require_resolved_identity()
    registration = self._identity_tasks.get(fingerprint)
    if registration is None:
        registration = asyncio.create_task(self._register_identity())
        registration.add_done_callback(self._consume_task_exception)
        self._identity_tasks[fingerprint] = registration
    await self._await_owned_future(registration)
    return _fingerprint_store_item(self, work_item)

    @classmethod
async def read_results(
    cls,
    path: str | Path,
    *,
    successes_only: bool = False,
    output_decoder: ValueDecoder | None = None,
    context_decoder: ValueDecoder | None = None,
    read_batch_size: int = 1000,
) -> BatchResult[Any, Any]:
    """Materialize a finite snapshot through an operationally read-only connection."""
    artifact_path = Path(path)
    page_size = _validate_positive_integer("read_batch_size", read_batch_size)
    executor = ThreadPoolExecutor(
        max_workers=1,
        thread_name_prefix="llm-minder-sqlite-reader",
    )
    loop = asyncio.get_running_loop()
    future = loop.run_in_executor(
        executor,
        cls._read_results_sync,
        artifact_path,
        successes_only,
        output_decoder,
        context_decoder,
        page_size,
    )
    try:
        return cast(BatchResult[Any, Any], await _await_without_cancelling(future))
    except asyncio.CancelledError:
        # The SQLite connection belongs to the worker thread. Let its
        # operation reach the helper's finally block before propagating
        # cancellation when the caller allows cleanup to finish. A repeated
        # cancellation remains prompt: the executor is then told to shut
        # down without synchronously blocking the event loop.
        try:
            await _await_without_cancelling(future)
        except asyncio.CancelledError:
            pass
        except Exception:
            # Cancellation remains the public outcome, but retrieving the
            # worker exception prevents an unobserved background failure.
            pass
        raise
    finally:
        if not future.done():
            future.add_done_callback(cls._consume_future_exception)
        executor.shutdown(wait=future.done())
